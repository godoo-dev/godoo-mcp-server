    test_tools_list(self, mcp_client_factory, odoo_harness):
      client = mcp_client_factory(url=odoo_harness.url, database=odoo_harness._database, username="admin", password="admin")
      async with client:
          resp = await client.post("/mcp/", json={"jsonrpc": "2.0", "method": "tools/list", "params": {}, "id": 1})
      assert resp.status_code == 200
      data = resp.json()
      assert "result" in data or "tools" in data  # MCP response shape
      tool_names = [t["name"] for t in (data.get("result", {}).get("tools", data.get("tools", [])))]
      assert "odoo_whoami" in tool_names

    test_full_vertical(self, mcp_client_factory, odoo_harness):
      client = mcp_client_factory(url=odoo_harness.url, database=odoo_harness._database, username="admin", password="admin")
      async with client:
          resp = await client.post("/mcp/", json={"jsonrpc": "2.0", "method": "tools/call", "params": {"name": "odoo_whoami", "arguments": {}}, "id": 2})
      assert resp.status_code == 200
      data = resp.json()
      # MCP tool call response: result.content[0].text contains the JSON output
      result_text = data["result"]["content"][0]["text"]
      import json
      result = json.loads(result_text)
      assert "uid" in result
      assert isinstance(result["uid"], int)
      assert result["uid"] > 0
      assert "db" in result

    test_concurrent_isolation(self, mcp_client_factory, odoo_harness):

      # Fire two concurrent calls via asyncio.gather
      import asyncio
      async def call_whoami(client_factory_kwargs: dict) -> dict:
          client = mcp_client_factory(**client_factory_kwargs)
          async with client:
              resp = await client.post("/mcp/", json={"jsonrpc": "2.0", "method": "tools/call", "params": {"name": "odoo_whoami", "arguments": {}}, "id": 1})
